CLASS Transaction<T>  
	ATTRIBUTE paymentMethod  

	CONSTRUCTOR(paymentMethod)  
		SET this.paymentMethod = paymentMethod  

	METHOD process()  
		PRINT "Processing payment with: " + paymentMethod  

CLASS UPI
CLASS Card
CLASS NetBanking

MAIN
    CREATE Transaction<UPI>
    CREATE Transaction<Card>
    CREATE Transaction<NetBanking>

    CALL process() on each
